================================================================================
EDITING - labels in SET, UPDATE and PROMPT-FOR
================================================================================

SET qty conv-loop: EDITING: READKEY. END.
UPDATE qty update-loop: EDITING: BELL. END.
PROMPT-FOR qty prompt-loop: EDITING: BELL. END.

--------------------------------------------------------------------------------

(source_code
  (set_statement
    (field field: (identifier))
    (editing_phrase label: (identifier) (readkey_statement)))
  (update_statement
    (field field: (identifier))
    (editing_phrase label: (identifier) (bell_statement)))
  (prompt_for_statement
    (field field: (identifier))
    (editing_phrase label: (identifier) (bell_statement))))

================================================================================
EDITING - labels after empty and populated frame phrases
================================================================================

SET qty WITH edit-loop: EDITING: READKEY. END.
UPDATE qty WITH rows DOWN edit-loop: EDITING: READKEY. END.
PROMPT-FOR qty WITH FRAME f WITH NO-VALIDATE edit-loop: EDITING: READKEY. END.

--------------------------------------------------------------------------------

(source_code
  (set_statement
    (field field: (identifier))
    (frame_phrase)
    (editing_phrase label: (identifier) (readkey_statement)))
  (update_statement
    (field field: (identifier))
    (frame_phrase
      (down value: (identifier)))
    (editing_phrase label: (identifier) (readkey_statement)))
  (prompt_for_statement
    (field field: (identifier))
    (frame_phrase frame: (identifier) (no_validate))
    (editing_phrase label: (identifier) (readkey_statement))))
