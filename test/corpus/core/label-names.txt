================================================================================
Labels - nonstandard names in control-flow clauses
================================================================================

1a: REPEAT ON ERROR UNDO 1a, RETRY 1a:
  UNDO 1a, NEXT 1a.
END.
$loop: REPEAT ON ENDKEY UNDO $loop, LEAVE $loop:
  NEXT $loop.
END.
-looper: REPEAT ON STOP UNDO -looper, RETRY -looper:
  UNDO -looper, RETRY -looper.
END.
quit/loop: REPEAT ON QUIT UNDO quit/loop, LEAVE quit/loop:
  LEAVE quit/loop.
END.

--------------------------------------------------------------------------------

(source_code
  (repeat_statement label: (identifier)
    (on_error_phrase undo_label: (identifier) retry_label: (identifier))
    (body (undo_statement undo_label: (identifier) next_label: (identifier))))
  (repeat_statement label: (identifier)
    (on_endkey_phrase undo_label: (identifier) leave_label: (identifier))
    (body (next_statement label: (identifier))))
  (repeat_statement label: (identifier)
    (on_stop_phrase undo_label: (identifier) retry_label: (identifier))
    (body (undo_statement undo_label: (identifier) retry_label: (identifier))))
  (repeat_statement label: (identifier)
    (on_quit_phrase undo_label: (identifier) leave_label: (identifier))
    (body (leave_statement label: (identifier)))))
